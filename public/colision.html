Main Beam Shader Idea

{String.raw`uniform float time;
varying vec2 vUv;

void main() {
  vUv = uv;

  vec3 pos = position;

  float wave1 = sin(pos.x * 20.0 + time * 6.0) * 0.03;
  float wave2 = sin(pos.x * 55.0 + time * 12.0) * 0.01;

  pos.y += wave1 + wave2;
  pos.z += cos(pos.x * 25.0 + time * 5.0) * 0.02;

  gl_Position = projectionMatrix * modelViewMatrix * vec4(pos, 1.0);
}`}

Fragment Glow

{String.raw`uniform vec3 color;
uniform float time;

varying vec2 vUv;

void main() {
  float glow = 1.0 - abs(vUv.y - 0.5) * 2.0;

  glow = pow(glow, 3.0);

  float pulse = 0.8 + sin(time * 8.0) * 0.2;

  vec3 finalColor = color * glow * pulse;

  gl_FragColor = vec4(finalColor, glow);
}`}
